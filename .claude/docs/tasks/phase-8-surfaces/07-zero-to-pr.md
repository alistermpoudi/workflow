# Phase 5 — Tâche 5.5 : ZeroPR.py (Zero-to-PR)

## Objectif

Fermer la boucle complète du cycle de développement. Après qu'une tâche est validée par `ExecutionLoop`, `ZeroPR` crée automatiquement un **commit conventionnel**, pousse la branche, et ouvre une **PR GitHub/GitLab avec description auto-générée** depuis le contenu de la tâche.

L'utilisateur ne touche plus jamais à git pour les tâches routinières — il ne valide que les PRs qui demandent une décision humaine.

## Dépendances

- Phase 1 (`GitManager`) ✅
- Phase 4 (MCP Server) ✅
- `httpx` pour l'API GitHub/GitLab

## Fichiers à Créer

- `src/workflow/tools/zero_pr.py` [CRÉER]

## Format du Commit Conventionnel

```
feat(TASK-007): authentification JWT avec refresh tokens

- Implémente POST /auth/login → access_token + refresh_token
- Middleware jwt_required sur les routes protégées
- Tests unitaires : 8 tests, coverage 94%

Auto-generated by Workflow
```

## Implémentation

```python
# src/workflow/tools/zero_pr.py
import json
import re
from pathlib import Path

import httpx

from workflow.core.project_memory import ProjectMemory
from workflow.tools.git_manager import GitManager
from workflow.llm.llm_provider import LLMProvider

PR_DESCRIPTION_PROMPT = """Génère une description de Pull Request professionnelle pour cette tâche.

Tâche :
{task_json}

Fichiers modifiés : {files}
Décisions techniques pertinentes : {decisions}

Format EXACT (Markdown) :
## {task_id} — {title}

### User Story
[user story de la tâche]

### Changements
[liste des fichiers créés/modifiés avec leur rôle]

### Critères d'acceptation
[liste avec [x] pour chaque critère]

### Décisions techniques
[décisions pertinentes du decisions.log]

---
🤖 *Généré automatiquement par Workflow*

Retourne UNIQUEMENT le Markdown."""


class ZeroPR:
    def __init__(self, project_root: str, llm: LLMProvider):
        self.project_root = Path(project_root).resolve()
        self.memory = ProjectMemory(project_root)
        self.git = GitManager(project_root)
        self.llm = llm

    async def commit_and_pr(
        self,
        task: dict,
        relevant_decisions: list[dict],
        github_token: str | None = None,
        gitlab_token: str | None = None,
        reviewers: list[str] | None = None,
    ) -> dict:
        """
        Créer le commit + pousser + ouvrir la PR.
        Retourne {'ok': True, 'pr_url': str}.
        """
        task_id = task['id']
        title = task.get('title', '')

        # 1. Commit conventionnel
        commit_msg = self._build_commit_message(task)
        await self.git.add_all()
        commit_result = await self.git.commit(commit_msg)
        if commit_result['exit_code'] != 0:
            return {'ok': False, 'error': f'Commit échoué : {commit_result["output"]}'}

        # 2. Pousser la branche
        current_branch = await self.git.current_branch()
        push_result = await self.git.push(current_branch)
        if push_result['exit_code'] != 0:
            return {'ok': False, 'error': f'Push échoué : {push_result["output"]}'}

        # 3. Générer la description de PR
        files_text = ', '.join(
            f.get('path', str(f)) if isinstance(f, dict) else str(f)
            for f in (task.get('files') or [])
        )
        decisions_text = '\n'.join(
            f'- {d.get("summary", "")}' for d in (relevant_decisions or [])[:5]
        )
        pr_body = await self.llm.ask(
            PR_DESCRIPTION_PROMPT.format(
                task_json=json.dumps(task, ensure_ascii=False, indent=2)[:1500],
                task_id=task_id,
                title=title,
                files=files_text or '(aucun fichier spécifié)',
                decisions=decisions_text or 'aucune',
            ),
            role='fast',
        )

        # 4. Créer la PR selon la plateforme
        project = await self.memory.get_project()
        repo_url = (project or {}).get('repo_url', '')

        if github_token and 'github.com' in repo_url:
            return await self._create_github_pr(
                task_id, title, current_branch, pr_body, github_token, repo_url, reviewers
            )
        elif gitlab_token and 'gitlab' in repo_url:
            return await self._create_gitlab_mr(
                task_id, title, current_branch, pr_body, gitlab_token, repo_url, reviewers
            )
        else:
            # Pas de token configuré → retourner le commit et le message de PR
            return {
                'ok': True,
                'committed': True,
                'pr_url': None,
                'pr_body': pr_body,
                'message': 'PR non créée : configure repo_url + token dans project.json',
            }

    def _build_commit_message(self, task: dict) -> str:
        task_id = task['id']
        title = task.get('title', '')
        files = task.get('files') or []

        # Type de commit depuis le titre
        title_lower = title.lower()
        if any(w in title_lower for w in ('fix', 'correc', 'bug', 'répar')):
            commit_type = 'fix'
        elif any(w in title_lower for w in ('refactor', 'nettoy', 'clean')):
            commit_type = 'refactor'
        elif any(w in title_lower for w in ('test', 'spec')):
            commit_type = 'test'
        elif any(w in title_lower for w in ('doc', 'readme', 'commentaire')):
            commit_type = 'docs'
        elif any(w in title_lower for w in ('setup', 'config', 'init', 'install')):
            commit_type = 'chore'
        else:
            commit_type = 'feat'

        scope = task_id.lower()
        changes = '\n'.join(
            f'- {f.get("path", str(f))} [{f.get("action", "modifié")}]'
            if isinstance(f, dict) else f'- {f}'
            for f in files[:5]
        )

        return (
            f'{commit_type}({scope}): {title}\n\n'
            f'{changes}\n\n'
            f'Closes {task_id}\n'
            f'Auto-generated by Workflow'
        )

    async def _create_github_pr(
        self,
        task_id: str,
        title: str,
        branch: str,
        body: str,
        token: str,
        repo_url: str,
        reviewers: list[str] | None,
    ) -> dict:
        # Extraire owner/repo depuis l'URL
        match = re.search(r'github\.com[:/](.+?)/(.+?)(?:\.git)?$', repo_url)
        if not match:
            return {'ok': False, 'error': f'URL GitHub non reconnue : {repo_url}'}

        owner, repo = match.group(1), match.group(2)
        api_url = f'https://api.github.com/repos/{owner}/{repo}/pulls'

        payload: dict = {
            'title': f'[{task_id}] {title}',
            'body': body,
            'head': branch,
            'base': 'main',
        }

        async with httpx.AsyncClient() as client:
            resp = await client.post(
                api_url,
                json=payload,
                headers={
                    'Authorization': f'Bearer {token}',
                    'Accept': 'application/vnd.github.v3+json',
                },
                timeout=30,
            )

        if resp.status_code not in (200, 201):
            return {'ok': False, 'error': f'GitHub API {resp.status_code} : {resp.text[:200]}'}

        pr_data = resp.json()
        pr_url = pr_data.get('html_url', '')

        # Assigner les reviewers si fournis
        if reviewers and pr_data.get('number'):
            await self._add_github_reviewers(
                owner, repo, pr_data['number'], reviewers, token
            )

        return {'ok': True, 'pr_url': pr_url, 'pr_number': pr_data.get('number')}

    async def _add_github_reviewers(
        self, owner: str, repo: str, pr_number: int, reviewers: list[str], token: str
    ) -> None:
        async with httpx.AsyncClient() as client:
            await client.post(
                f'https://api.github.com/repos/{owner}/{repo}/pulls/{pr_number}/requested_reviewers',
                json={'reviewers': reviewers},
                headers={'Authorization': f'Bearer {token}'},
                timeout=15,
            )

    async def _create_gitlab_mr(
        self,
        task_id: str,
        title: str,
        branch: str,
        body: str,
        token: str,
        repo_url: str,
        reviewers: list[str] | None,
    ) -> dict:
        # Extraire le project path depuis l'URL GitLab
        match = re.search(r'gitlab[^/]*/(.+?)(?:\.git)?$', repo_url)
        if not match:
            return {'ok': False, 'error': f'URL GitLab non reconnue : {repo_url}'}

        from urllib.parse import quote
        project_path = quote(match.group(1), safe='')
        gitlab_host = re.search(r'(https?://[^/]+)', repo_url)
        base = gitlab_host.group(1) if gitlab_host else 'https://gitlab.com'
        api_url = f'{base}/api/v4/projects/{project_path}/merge_requests'

        payload = {
            'title': f'[{task_id}] {title}',
            'description': body,
            'source_branch': branch,
            'target_branch': 'main',
        }

        async with httpx.AsyncClient() as client:
            resp = await client.post(
                api_url,
                json=payload,
                headers={'PRIVATE-TOKEN': token},
                timeout=30,
            )

        if resp.status_code not in (200, 201):
            return {'ok': False, 'error': f'GitLab API {resp.status_code} : {resp.text[:200]}'}

        mr_data = resp.json()
        return {'ok': True, 'pr_url': mr_data.get('web_url', '')}
```
